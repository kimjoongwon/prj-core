## 역할

파일 업로드를 위한 통합 패널 위젯입니다. DropZone과 UploadQueue를 조합하여 파일 선택부터 업로드 진행까지의 전체 UX를 제공합니다.

## Props

```typescript
interface UploadPanelProps {
  onFilesSelected: (files: File[]) => void;  // 파일 선택 핸들러
  onCancelAll?: () => void;                  // 전체 취소 핸들러
  onRetryFailed?: () => void;                // 실패 재시도 핸들러
  onComplete?: () => void;                   // 완료 핸들러
  onUploadMore?: () => void;                 // 더 업로드 핸들러
  queueItems?: UploadQueueItem[];            // 업로드 큐 항목
  accept?: string[];                         // 허용 파일 타입
  maxFileSize?: number;                      // 최대 파일 크기
  maxFiles?: number;                         // 최대 파일 수
  multiple?: boolean;                        // 다중 파일 허용
  showQueue?: boolean;                       // 큐 표시 여부
  status?: 'idle' | 'uploading' | 'completed' | 'failed'; // 패널 상태
  completedCount?: number;                   // 완료된 파일 수
  failedCount?: number;                      // 실패한 파일 수
  className?: string;                        // 추가 클래스
}
```
