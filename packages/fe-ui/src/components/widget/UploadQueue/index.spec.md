## Props

```typescript
interface UploadQueueProps {
  items: UploadQueueItem[];                  // 업로드 항목 목록
  onCancelAll?: () => void;                  // 전체 취소 핸들러
  onRetryFailed?: () => void;                // 실패 항목 재시도 핸들러
  onItemPause?: (id: string) => void;        // 개별 일시정지 핸들러
  onItemResume?: (id: string) => void;       // 개별 재개 핸들러
  onItemCancel?: (id: string) => void;       // 개별 취소 핸들러
  onItemRetry?: (id: string) => void;        // 개별 재시도 핸들러
  maxHeight?: string | number;               // 최대 높이
  emptyMessage?: string;                     // 빈 상태 메시지
  className?: string;                        // 추가 클래스
}

interface UploadQueueItem {
  id: string;
  file: File;
  fileName: string;
  fileSize: number;
  progress?: number;
  status: 'pending' | 'uploading' | 'paused' | 'completed' | 'failed';
  errorMessage?: string;
}
```
